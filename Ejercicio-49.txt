Algoritmo ContarNumeroEnArreglo
    Definir n, i, buscado, contador Como Entero
    Dimension arreglo[100]

    Escribir "¿Cuántos elementos tendrá el arreglo?"
    Leer n

    Para i <- 1 Hasta n Hacer
        Escribir "Ingrese el elemento ", i, ":"
        Leer arreglo[i]
    FinPara

    Escribir "¿Qué número desea buscar?"
    Leer buscado

    contador <- 0

    Para i <- 1 Hasta n Hacer
        Si arreglo[i] = buscado Entonces
            contador <- contador + 1
        FinSi
    FinPara

    Escribir "El número ", buscado, " aparece ", contador, " vez/veces en el arreglo."
FinAlgoritmo
